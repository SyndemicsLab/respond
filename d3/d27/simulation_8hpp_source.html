167 void Run(int duration = -1) {
168 if (duration > 0) {
169 _duration = duration;
170 }
171 LogInfo(_log_name, "Running simulation for duration of " +
172 std::to_string(_duration) + " timesteps.");
173 for (const auto &model : _models) {
174 model->SetFinalTimestep(_duration);
175 model->RunTimesteps();
176 }
177 }

32enum class HistoryMode : int {

34 kAccumulated = 1, // Accumulated contributions over timesteps

42inline HistoryMode GetDefaultHistoryMode(const std::string &name) {
43 if (name == "intervention_admission" || name == "total_overdose" ||
44 name == "fatal_overdose" || name == "background_death") {
45 return HistoryMode::kAccumulated;
46 }
47 return HistoryMode::kSnapshot;
48}

202 void AddState(const Eigen::Ref<const Eigen::VectorXd> &state,
203 int timestep = -1) {
204 if (timestep < 0) {
205 timestep = GetNextTimestep();
206 }
207
208 const auto insertion_point =
209 std::lower_bound(_timesteps.begin(), _timesteps.end(), timestep);
210 const auto index =
211 static_cast<size_t>(insertion_point - _timesteps.begin());
212 if (insertion_point != _timesteps.end() &&
213 *insertion_point == timestep) {
214 _states[index] = state;
215 return;
216 }
217
218 _timesteps.insert(insertion_point, timestep);
219 _states.insert(_states.begin() + static_cast<std::ptrdiff_t>(index),
220 state);
221 }
222
225 void AccumulateState(const Eigen::Ref<const Eigen::VectorXd> &state) {
226 if (_mode != HistoryMode::kAccumulated) {
227 LogWarning(_log_name, "AccumulateState called on non-accumulated "
228 "history, adding state instead: " +
229 _name);
230 AddState(state);
231 return;
232 }
233
234 if (_pending_state.size() == 0) {
235 _pending_state = state;
236 return;
237 }
238 _pending_state += state;
239 }
240
244 void FlushPendingState(int timestep, Eigen::Index state_size) {
245 if (_mode != HistoryMode::kAccumulated) {
246 LogInfo(_log_name,
247 "FlushPendingState called on non-accumulated history, "
248 "no pending state to flush: " +
249 _name);
250 return;
251 }
252
253 Eigen::VectorXd value;
254 if (_pending_state.size() > 0) {
255 value = _pending_state;
256 } else {
257 LogInfo(_log_name,
258 "FlushPendingState called with no pending state, "
259 "recording zero vector: " +
260 _name);
261 value = Eigen::VectorXd::Zero(state_size);
262 }
263
264 AddState(value, timestep);
265 _pending_state.resize(0);
266 }

333 std::vector<Eigen::VectorXd> GetStateAsVector() const {
334 std::vector<Eigen::VectorXd> ret;
335 if (_states.empty()) {
336 LogWarning(_log_name,
337 "GetStateAsVector called on empty history: " + _name);
338 return {};
339 }
340 int default_size = _states.front().size();
341 int tstep = 0;
342 for (size_t index = 0; index < _timesteps.size(); ++index) {
343 const int recorded_timestep = _timesteps[index];
344 const auto &recorded_state = _states[index];
345 while (recorded_timestep > tstep) {
346 ret.push_back(GetZeroVector(default_size));
347 tstep++;
348 }
349 ret.push_back(recorded_state);
350 tstep++;
351 }
352 return ret;
353 }

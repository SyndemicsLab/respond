61 TransitionSlotProxy &operator=(const Transition &transition) {
62 _owner->GetTransitionRefOrThrow(_idx);

76 "Error attempting to assign null model pointer.");

78 _owner->GetTransitionRefOrThrow(_idx);
79 _owner->_transitions[_idx] = transition->clone();
80 return *this;
81 }

219 std::unique_ptr<Transition> RemoveTransition(size_t idx) {
220 if (idx >= _transitions.size()) {
221 LogWarning(_log_name, "Index out of range in RemoveTransition: " +
222 std::to_string(idx));

224 "Error attempting to RemoveTransition by index.");

227 _transitions.erase(_transitions.begin() + idx);

382 bool operator==(const Timestep &other) const {
383 if (_transitions.size() != other._transitions.size()) {
384 return false;
385 }
386 for (size_t i = 0; i < _transitions.size(); ++i) {
387 if (_transitions[i]->GetName() !=
388 other._transitions[i]->GetName()) {
389 return false;
390 }
391 if (_transitions[i]->GetMatrices().size() !=
392 other._transitions[i]->GetMatrices().size()) {
393 return false;
394 }
395 for (size_t j = 0; j < _transitions[i]->GetMatrices().size(); ++j) {
396 if (!_transitions[i]->GetMatrices()[j].isApprox(
397 other._transitions[i]->GetMatrices()[j])) {
398 return false;
399 }
400 }
401 }
402 return true;
403 }
